cmake_minimum_required(VERSION 3.20)
project(PmacSdkExamples LANGUAGES C CXX)
include(CTest)
# 本工程仅引用已编译库，不调用 Cargo；也可复制本目录到自己的项目中。
get_filename_component(_default_sdk "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(PMAC_SDK_ROOT "${_default_sdk}" CACHE PATH "Extracted PMAC SDK root")
find_package(Pmac CONFIG REQUIRED PATHS "${PMAC_SDK_ROOT}/cmake" NO_DEFAULT_PATH)
add_executable(sdk_c c/sdk_smoke.c)
add_executable(sdk_cpp cpp/sdk_smoke.cpp)
add_executable(sdk_cpp98 cpp/sdk_smoke.cpp)
target_compile_definitions(sdk_cpp98 PRIVATE PMAC_EXAMPLE_CPP98)
add_executable(getresponse_c c/getresponse.c)
add_executable(getresponse_cpp cpp/getresponse.cpp)
add_executable(geterrors_c c/geterrors.c)
add_executable(geterrors_cpp cpp/geterrors.cpp)
add_executable(gather_cpp cpp/gather.cpp)
set_target_properties(sdk_c getresponse_c geterrors_c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
set_target_properties(sdk_cpp getresponse_cpp geterrors_cpp gather_cpp PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
set_target_properties(sdk_cpp98 PROPERTIES CXX_STANDARD 98 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
foreach(example sdk_c sdk_cpp sdk_cpp98 getresponse_c getresponse_cpp geterrors_c geterrors_cpp gather_cpp)
    target_link_libraries(${example} PRIVATE Pmac::pmac)
    if(MSVC)
        # 中文源码按 UTF-8 读取，不依赖开发机的系统代码页。
        target_compile_options(${example} PRIVATE /W4 /WX /utf-8)
        target_compile_definitions(${example} PRIVATE _CRT_SECURE_NO_WARNINGS)
        add_custom_command(TARGET ${example} POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:Pmac::pmac>" "$<TARGET_FILE_DIR:${example}>" VERBATIM)
    else()
        target_compile_options(${example} PRIVATE -Wall -Wextra -Werror)
        # 构建树自动获得导入库路径；保留相对路径供移动可执行文件时使用。
        set_target_properties(${example} PROPERTIES INSTALL_RPATH "$ORIGIN/../native")
    endif()
endforeach()
if(BUILD_TESTING)
    # 测试不读 PMAC_HOST，也不发送网络/运动命令。
    add_test(NAME sdk_c COMMAND sdk_c)
    add_test(NAME sdk_cpp COMMAND sdk_cpp)
    add_test(NAME sdk_cpp98 COMMAND sdk_cpp98)
    add_test(NAME getresponse_c_check COMMAND getresponse_c --check)
    add_test(NAME getresponse_cpp_check COMMAND getresponse_cpp --check)
endif()
